# Colors
tema	toggle pywal <-> default theme (or Super+Shift+T)
tema image.jpg	new colors + wallpaper from an image
tema on	pywal colors + last pywal wallpaper
tema off	default colors + plain dark wallpaper
tema status	which theme is active

# Disks and displays
dmenu_mount	mount / unmount USB drives (Super+Shift+M)
dmenu_display	displays (Super+Shift+D)
cripto_pendrive.sh open /dev/sdb1	open encrypted USB drive
cripto_pendrive.sh close	close encrypted USB drive

# Other
layout_change.sh	keyboard ABNT2 <-> US international (Super+Alt+K)
conserv_mode.sh	battery conservation mode
camera_cell	phone as a webcam
download_music.sh URL album	download an album (yt-dlp)
pomodoro	pomodoro timer
cheats --check	check that the cheat sheets match config.h

# Behind the shortcuts (can also be run by hand)
dmenu_power	power menu (Super+Shift+Q)
dmenu_emoji	emoji and symbol picker (Super+Shift+E)
dmenu_run	program launcher with history (Super+P)
cheats --menu	this cheat sheet in dmenu (Super+F1)
screenshot_satty.sh / screenshot_satty.sh -e	screenshot / with editor (Super+Shift+A / S)
controle.sh vol_up / vol_down / vol_mute	volume with notification (media keys, max 100%)
controle.sh vol_limit	bring volume back to 100% if above (runs at login)
controle.sh bright_up / bright_down	brightness with notification (media keys)
st-urlhandler open / copy	links on screen in st (Alt+L / Alt+Y)

# Older scripts
duplicate_terminal.sh	new st in the current directory (st has Ctrl+Shift+Enter now)
arrumar_monitor.sh	HDMI left of the laptop (dmenu_display does this and more)
screenshot.sh / screenshot_flameshot.sh	older screenshot variants (maim / flameshot)
mesa_config.sh	Wacom tablet mapping (runs at login)
walupdate.sh	old name of "tema"
